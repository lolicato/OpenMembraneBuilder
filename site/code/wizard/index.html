    Wizard¶

    wizard ¶

    The wizard: the step-by-step workflow of the graphical interface.

    • state – what the wizard remembers (chosen workflow, current page).
    • navigation – moving forward and back.
    • page – the welcome page, the hand-over between builders and saving config.json.
    • gui – the results page (download and 3D view).
    • run_simulations – the "Run simulations" button of the results page.
    • runner – the same build without the interface, from a config.json (omb cli).

    state ¶

    What the wizard remembers between clicks: the workflow you chose and where you are in it.

    Streamlit redraws the whole page after every click. The wizard state is kept in Streamlit's session memory, so the wizard knows which page to draw next.

    def go_forward(state: WizardState) -> None:
    """Move the wizard to the next step of the chosen workflow.

    From the welcome page this is the first step of the workflow; after the last step it is
    the results page.

    Args:
        state: The wizard's state (see `WizardState`), changed in place.
    """
    if state.step == WizardStep.SELECT_PATHWAY:
        state.step_index = 0
        _sync_step(state)
        return

    state.step_index += 1
    if state.step_index >= len(state.pathway.steps):
        state.step = WizardStep.RESULTS
    else:
        _sync_step(state)

    def go_back(state: WizardState) -> None:
    """Move the wizard one step back.

    From the first step of a workflow this returns to the welcome page, where another
    workflow can be chosen.

    Args:
        state: The wizard's state (see `WizardState`), changed in place.
    """
    if state.step_index > 0:
        state.step_index -= 1
        _sync_step(state)
        return

    # Already on the first step of the pathway -> return to the pathway selection
    # (previously this was a silent no-op and the user was stuck).
    state.step = WizardStep.SELECT_PATHWAY
    state.step_index = 0
    state.locked = False

    page ¶

    The wizard: the page that guides you from the welcome screen to the finished systems.

    Depending on the workflow chosen on the welcome page, the wizard shows

    1. the protein preparation (only with proteins),
    2. the membrane builder (composition, box, solvation, protein placement, preview),
    3. the results (download and 3D view).

    When the build is done, the settings of all steps are written to config.json in the output folder (and added to the zip archive), so the same systems can be rebuilt later with omb cli config.json.

    def _json_fallback(obj):
    """Turn values that JSON cannot store directly (objects, sets, …) into storable ones.

    Args:
        obj: The value that could not be written.

    Returns:
        A dictionary, list or text version of the value.
    """
    if hasattr(obj, "__dict__"):
        return vars(obj)
    if hasattr(obj, "__iter__"):
        return list(obj)
    return str(obj)

    def _save_full_config(config_map: dict, output_folder: str) -> str:
    """Write the settings of all wizard steps to `<output>/config.json`.

    Values that only matter during a session (temporary folders, intermediate text) are
    left out, so the file describes the build and nothing else.

    Args:
        config_map: The finished settings of each step, e.g.
            `{"martinize": ..., "membrane": ...}`.
        output_folder: The output folder of the build.

    Returns:
        The path of the written `config.json`.
    """
    combined = OrderedDict()
    for key in _ORDER:
        if key not in config_map:
            continue
        cfg = config_map[key]
        data = asdict(cfg) if is_dataclass(cfg) else vars(cfg).copy()
        for k in _STRIP_KEYS:
            data.pop(k, None)
        combined[key] = data

    os.makedirs(output_folder, exist_ok=True)
    path = os.path.join(output_folder, "config.json")
    with open(path, "w") as f:
        json.dump(combined, f, indent=4, default=_json_fallback)
    return path

    def render_wizard():
    """Draw the page that belongs to the current wizard step.

    This is called on every redraw of the browser page (`app.py`). Once a workflow has
    been started, the sidebar also shows a "Change workflow" button that starts over.
    """
    state = get_state()

    if state.locked:
        with st.sidebar:
            if st.button("↩ Change workflow", use_container_width=True):
                st.session_state.clear()
                st.rerun()

    if state.step == WizardStep.SELECT_PATHWAY:
        _select_pathway(state)

    elif state.step == WizardStep.MARTINIZE:
        from openmembranebuilder.builders.martinize.page import main as martinize_main

        result, cfg = martinize_main(temp_folder=_ensure_temp_folder(state))
        if result == "done" and cfg:
            st.session_state.setdefault("completed_configs", {})["martinize"] = cfg
            go_forward(state)
            st.rerun()
        elif result == "back":
            go_back(state); st.rerun()

    elif state.step == WizardStep.MEMBRANE:
        _render_membrane_stage(state)

    elif state.step == WizardStep.RESULTS:
        from openmembranebuilder.wizard.gui import render_wizard_results

        membrane_cfg = st.session_state.get("completed_configs", {}).get("membrane")
        box_nm = None
        if membrane_cfg is not None:
            box_nm = (
                float(membrane_cfg.box_x),
                float(membrane_cfg.box_y),
                float(membrane_cfg.box_z),
            )

        render_wizard_results(
            st.session_state.get("output_pdbs", []),
            st.session_state.get("zip_path"),
            box_nm=box_nm,
            output_folder=getattr(membrane_cfg, "base_folder", None),
        )

    def _render_membrane_stage(state) -> None:
    """Show the membrane builder and finish the build when it reports "done".

    After the build, the full configuration is saved and added to the zip archive, the
    upload folder is removed and the wizard moves on to the results page.

    Args:
        state: The wizard state.
    """
    completed = st.session_state.setdefault("completed_configs", {})

    from openmembranebuilder.builders.membrane.page import main as membrane_main




    result, cfg = membrane_main(
        GlobalInfo(),
        has_protein=state.has_protein,
        temp_folder=_ensure_temp_folder(state),
        protein_only=state.pathway.protein_only if state.pathway else False
    )

    if result == "done" and cfg:
        completed["membrane"] = cfg
        # output_pdbs / zip_path are already stored in session_state by the membrane page
        config_path = _save_full_config(completed, cfg.base_folder)
        st.session_state["final_config_path"] = config_path
        # the zip was made by the runner before config.json existed: add it
        from openmembranebuilder.core.filemanager import FileManager
        FileManager.add_to_zip(st.session_state.get("zip_path"), config_path, "config.json")
        _cleanup_temp_folder(state)
        go_forward(state)
        st.rerun()

    elif result == "back":
        go_back(state)
        st.rerun()

    # Note: never reached – the membrane page ends the page drawing itself (st.stop),
    # and `group_labels` does not exist. Left over from an earlier version.
    selected_group = st.radio(
        "What do you want to build?",
        group_labels,
        index=group_labels.index("Membrane only (no protein)"),  # ← default
    )

    def _select_pathway(state) -> None:
    """The welcome page: choose what to build and press "Start".

    Starting clears the results of an earlier run in the same browser session.

    Args:
        state: The wizard state.
    """
    st.title("Welcome")
    st.subheader("Choose your workflow")

    labels = [p.label for p in PATHWAYS]
    default_label = state.pathway.label if state.pathway else "Membrane only (no protein)"

    chosen_label = st.radio(
        "What do you want to build?",
        labels,
        index=labels.index(default_label),
    )
    selected = next(p for p in PATHWAYS if p.label == chosen_label)

    st.divider()


    if st.button("Start →", type="primary", use_container_width=True):
        # a fresh start must not inherit results of a previous run
        st.session_state.pop("completed_configs", None)
        st.session_state.pop("protein_files", None)
        state.pathway = selected
        state.locked = True
        go_forward(state)
        st.rerun()

    def _prepare_display_pdb(pdb_path: str, box_nm):
    """A copy of the system for the 3D view, with a split protein made whole again.

    Args:
        pdb_path: The built system (`system.pdb`).
        box_nm: Box size (x, y, z) in nm, needed to know how far to move the pieces.

    Returns:
        The path of a repaired copy in the temporary folder, or the original path when
        nothing needs (or can) be repaired.
    """
    if not pdb_path or box_nm is None or not os.path.exists(pdb_path):
        return pdb_path

    try:
        import MDAnalysis as mda

        universe = mda.Universe(pdb_path)

        protein = universe.select_atoms(
            "resname " + " ".join(sorted(_PROTEIN_RESNAMES))
        )

        if len(protein) == 0:
            return pdb_path

        box_lengths = np.asarray(box_nm, dtype=float) * 10.0
        universe.dimensions = np.asarray(
            [
                box_lengths[0],
                box_lengths[1],
                box_lengths[2],
                90.0,
                90.0,
                90.0,
            ],
            dtype=np.float32,
        )

        protein_positions = protein.positions.copy()
        original_span = protein_positions.max(axis=0) - protein_positions.min(axis=0)

        # If the protein is not split across a boundary, use the original PDB.
        if not np.any(original_span > 0.5 * box_lengths):
            return pdb_path

        repaired_positions = _repair_protein_periodic_split(
            protein_positions,
            box_lengths,
        )

        if repaired_positions is None:
            return pdb_path

        # Apply the repaired coordinates only to the protein atoms.
        protein.positions = repaired_positions

        session_dir = os.path.join(
            tempfile.gettempdir(),
            "openmembranebuilder-display",
        )
        os.makedirs(session_dir, exist_ok=True)

        display_path = os.path.join(
            session_dir,
            f"{os.path.basename(pdb_path)}-display-pbc-repaired.pdb",
        )

        with mda.Writer(display_path) as writer:
            writer.write(universe.atoms)

        return display_path

    except Exception as exc:
        st.warning(f"Could not repair PBC display for {pdb_path}: {exc}")
        return pdb_path

    def _repair_protein_periodic_split(protein_positions, box_lengths):
    """Join the pieces of a protein that is split across the box edge.

    Along each direction, the largest empty gap between protein beads is taken as the cut;
    the beads beyond it are moved back by one box length.

    Args:
        protein_positions: Protein bead positions in Å.
        box_lengths: Box lengths in Å.

    Returns:
        The repaired positions, or `None` when the protein still does not fit together.
    """
    repaired = protein_positions.copy()
    original_span = repaired.max(axis=0) - repaired.min(axis=0)

    for axis in range(3):
        span = original_span[axis]
        box_length = box_lengths[axis]

        # No likely PBC split in this direction.
        if span <= 0.5 * box_length:
            continue

        values = np.sort(repaired[:, axis])
        gaps = np.diff(values)

        if len(gaps) == 0:
            continue

        largest_gap_index = int(np.argmax(gaps))
        largest_gap = gaps[largest_gap_index]

        # The largest coordinate gap must plausibly be the periodic gap.
        if largest_gap <= 0.5 * box_length:
            continue

        split_value = values[largest_gap_index]

        # Atoms above the gap are shifted by -box length.
        candidate = repaired.copy()
        upper = candidate[:, axis] > split_value
        candidate[upper, axis] -= box_length

        candidate_span = candidate.max(axis=0) - candidate.min(axis=0)

        if candidate_span[axis] < span:
            repaired = candidate
            original_span = candidate_span

    if np.any(original_span > 0.5 * box_lengths):
        return None

    return repaired

    def running_pid(pid_file: str) -> int | None:
    """The process number of the simulations, if they are still running.

    Args:
        pid_file: The `run.pid` written when the simulations were started.

    Returns:
        The process number, or `None` when nothing was started or it has finished.
    """
    try:
        pid = int(open(pid_file).read().strip())
    except (OSError, ValueError):
        return None
    proc = _STARTED.get(pid)
    if proc is not None:  # started by this app: ask it directly
        return pid if proc.poll() is None else None
    try:  # started earlier (e.g. before the app was restarted)
        os.kill(pid, 0)
    except (OSError, ProcessLookupError):
        return None
    return pid

    def start_run(output_folder: str) -> int:
    """Start `run.sh` in the `systems` folder of a build, in the background.

    The script keeps running when the page is closed. Its output is added to
    `systems/run.log`; its process number is written to `systems/run.pid`.

    Args:
        output_folder: The output folder of the build.

    Returns:
        The process number of the started script.
    """
    systems, script, log_file, pid_file = _paths(output_folder)
    with open(log_file, "ab") as log:
        proc = subprocess.Popen(
            ["bash", os.path.basename(script)],
            cwd=systems,                       # run.sh runs every system folder in here
            stdout=log,
            stderr=subprocess.STDOUT,
            stdin=subprocess.DEVNULL,
            start_new_session=True,            # own process group: survives the page, can be stopped as a whole
        )
    _STARTED[proc.pid] = proc
    with open(pid_file, "w") as f:
        f.write(str(proc.pid))
    return proc.pid

    def render_run_controls(output_folder: str | None) -> None:
    """The "Run simulations" part of the results page.

    Shows **▶ Run simulations** while nothing runs, and the state, the latest lines of
    `run.log` and **⏹ Stop** while the simulations run. Nothing is shown when running is
    switched off (`omb gui --no-run`), or when the build has no `systems/run.sh`.

    The button is disabled, with an explanation, when `gmx` or `jq` cannot be found.

    Args:
        output_folder: The output folder of the build that was just made.
    """
    if not run_allowed() or not output_folder:
        return
    systems, script, log_file, pid_file = _paths(output_folder)
    if not os.path.exists(script):
        return

    st.subheader("Run the simulations")
    st.caption(
        f"Starts `run.sh` in `{systems}`: energy minimisation, equilibration (eq1–eq3) and "
        "production for every system, on this computer. It keeps running when you close the page."
    )

    pid = running_pid(pid_file)
    if pid is not None:
        st.info(f"⏳ The simulations are running (process {pid}).")
        left, right, _ = st.columns([1, 1, 4])
        if left.button("🔄 Refresh", use_container_width=True, key="run_refresh"):
            st.rerun()
        if right.button("⏹ Stop", use_container_width=True, key="run_stop"):
            stop_run(pid)
            st.rerun()
    else:
        missing = missing_programs()
        if missing:
            st.warning(
                f"Cannot run here: {', '.join(missing)} not found. Make GROMACS (`gmx`) and "
                "`jq` available, then restart `omb gui` – or run `systems/run.sh` yourself."
            )
        elif os.path.exists(log_file):
            st.success("The last run has ended – see the log below.")
        if st.button("▶ Run simulations", type="primary", disabled=bool(missing), key="run_start"):
            start_run(output_folder)
            st.rerun()

    log = _tail(log_file)
    if log:
        with st.expander("run.log (last lines)", expanded=pid is not None):
            st.code(log, language="text")

    def run_from_config(config_path: str):
    """Build the systems described in a `config.json`.

    A new output folder `output/OpenMembraneBuilder-<date>_<time>-headless` is created
    (unless the file names one), the systems are built in it, the configuration is copied
    next to them and added to the zip archive in `downloads/`.

    Args:
        config_path: Path of the `config.json`, usually one written by an earlier build.
    """
    with open(config_path) as f:
        data = json.load(f)

    # The saved membrane config already contains the RESULTS of the earlier builders
    # (protein_files = martinize outputs, molecule_builder_index = cglipid outputs),
    # and the temp folders those builders used no longer exist. Re-running them would
    # fail (or just duplicate work), so they are skipped.
    membrane = data.get("membrane", {})
    skip = set()
    if membrane.get("protein_files"):
        skip.add("martinize")
    if membrane.get("molecule_builder_index"):
        skip.add("cglipid")

    for builder_name, config_dict in data.items():
        if builder_name in skip:
            print(f"[headless] skipping '{builder_name}': its results are stored in the membrane config")
            continue

        config_module = importlib.import_module(f"openmembranebuilder.builders.{builder_name}.config")
        config = config_module.Config(**config_dict)

        if builder_name == "membrane" and not config.base_folder:
            from openmembranebuilder.core.global_info import GlobalInfo
            ts = datetime.now().strftime("%Y%m%d_%H%M%S")
            config.base_folder = os.path.join(GlobalInfo().base_folder, f"OpenMembraneBuilder-{ts}-headless")

        runner_class = _load_runner_class(builder_name)
        result = runner_class().run(config)

        # keep the configuration with the output and in the download zip, as the GUI does
        if builder_name == "membrane" and getattr(config, "base_folder", None):
            from openmembranebuilder.core.filemanager import FileManager
            out_cfg = os.path.join(config.base_folder, "config.json")
            if os.path.abspath(out_cfg) != os.path.abspath(config_path):
                with open(out_cfg, "w") as f:
                    json.dump(data, f, indent=4)
            if isinstance(result, tuple) and len(result) == 2:
                FileManager.add_to_zip(result[1], out_cfg, "config.json")
